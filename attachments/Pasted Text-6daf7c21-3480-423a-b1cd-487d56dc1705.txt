        function readJson(key, fallback) {
            try {
                const raw = localStorage.getItem(key);
                if (!raw) return fallback;
                const parsed = JSON.parse(raw);
                return parsed ?? fallback;
            } catch {
                return fallback;
            }
        }

        function writeJson(key, value) {
            try {
                localStorage.setItem(key, JSON.stringify(value));
                return true;
            } catch {
                showToast('تعذر حفظ البيانات محلياً');
                return false;
            }
        }

        function normalizePhone(value) {
            const digits = String(value || '').replace(/\D/g, '');
            if (/^0[^1_5679]\d{8}$/.test(digits)) {
                return `213${digits.slice(1)}`;
            }
            if (/^213[^1_5679]\d{8}$/.test(digits)) {
                return digits;
            }
            return '';
        }

        function isSafeImageUrl(value) {
            if (!value) return true;
            try {
                const url = new URL(value, window.location.origin);
                return url.protocol === 'https:' || (url.origin === window.location.origin && url.protocol === 'http:');
            } catch {
                return false;
            }
        }

        function normalizeProduct(p) {
            const price = Number(p.price);
            if (!p.id || !p.title || !Number.isInteger(price) || price <= 0) return null;
            return {
                id: String(p.id),
                title: String(p.title).trim().slice(0, 120),
                price,
                image: isSafeImageUrl(p.image) ? p.image : '',
                desc: String(p.desc || '').trim().slice(0, 1000),
                stock: Number.isInteger(Number(p.stock)) ? Math.max(0, Number(p.stock)) : 1,
                status: p.status === 'approved' ? 'approved' : 'pending',
                active: p.active !== false
            };
        }

        function normalizeStore(s) {
            return {
                id: String(s.id),
                name: String(s.name || '').trim().slice(0, 80),
                owner: String(s.owner || '').trim().slice(0, 80),
                phone: normalizePhone(s.phone),
                email: String(s.email || '').trim().slice(0, 120),
                avatar: isSafeImageUrl(s.avatar) ? s.avatar : '',
                status: s.status === 'approved' ? 'approved' : 'pending',
                active: Boolean(s.active),
                products: Array.isArray(s.products) ? s.products.map(normalizeProduct).filter(Boolean) : []
            };
        }

        function getPublicProducts(store) {
            return store.products.filter(p => p.active && p.status === 'approved' && p.stock > 0);
        }

        function navigate(viewId, params = {}) {
            ['home', 'stores', 'store-detail', 'merchant', 'product-direct'].forEach(v => {
                document.getElementById(`view-${v}`).classList.add('hidden');
            });
            document.getElementById(`view-${viewId}`).classList.remove('hidden');
            window.scrollTo({ top: 0, behavior: 'smooth' });

            const url = new URL(window.location.href);
            url.search = '';

            if (viewId === 'product-direct') {
                url.searchParams.set('store', params.storeId);
                url.searchParams.set('product', params.productId);
            } else if (viewId === 'store-detail') {
                url.searchParams.set('store', params.storeId);
            }

            history.pushState({ viewId, params }, '', url);

            if (viewId === 'home') renderFeaturedProducts();
            if (viewId === 'stores') renderStoresList();
            if (viewId === 'store-detail' && params.storeId) renderStoreDetail(params.storeId);
            if (viewId === 'merchant') checkMerchantState();
            if (viewId === 'product-direct' && params.storeId) renderDirectProduct(params.storeId, params.productId);
        }

        window.addEventListener('popstate', () => {
            routeFromUrl();
        });

        function routeFromUrl() {
            const params = new URLSearchParams(window.location.search);
            const storeId = params.get('store');
            const productId = params.get('product');

            if (storeId && productId) {
                navigate('product-direct', { storeId, productId });
            } else if (storeId) {
                navigate('store-detail', { storeId });
            } else {
                navigate('home');
            }
        }

        function renderFeaturedProducts() {
            const grid = document.getElementById('featuredProductsGrid');
            grid.replaceChildren();
            let count = 0;

            for (const store of getPublicStores()) {
                for (const product of getPublicProducts(store)) {
                    grid.appendChild(createProductCard(store, product));
                    count++;
                }
            }

            if (count === 0) {
                const empty = document.createElement('div');
                empty.className = "col-span-full text-center py-12 text-slate-400";
                empty.textContent = "لا توجد منتجات متاحة حالياً";
                grid.appendChild(empty);
            }
        }

        function checkMerchantState() {
            const authSec = document.getElementById('merchantAuthSection');
            const dashArea = document.getElementById('merchantDashboardArea');

        function createDemoStore() {
            const storeName = document.getElementById('merchantStoreName').value.trim();
            const owner = document.getElementById('merchantOwnerName').value.trim();
            const rawPhone = document.getElementById('merchantPhone').value.trim();
            const email = document.getElementById('merchantEmail').value.trim();

            const phone = normalizePhone(rawPhone);

            if (storeName.length < 3 || owner.length < 3) {
                showToast('يرجى ملء اسم المتجر واسم المالك بشكل صحيح');
                return;
            }

            if (!phone) {
                showToast('أدخل رقم هاتف جزائري صحيح (مثال: 0775554695)');
                return;
            }

            const newStore = {
                id: 'store-' + crypto.randomUUID(),
                name: storeName,
                owner: owner,
                phone: phone,
                email: email || 'merada35@gmail.com',
                avatar: 'https://images.unsplash.com/photo-1472851294608-062f824d29cc?auto=format&fit=crop&w=400&q=80',
                status: 'approved',
                active: true,
                products: []
            };

            dbStores.push(newStore);
            currentMerchantStore = newStore;

            writeJson(STORAGE_KEYS.stores, dbStores);
            writeJson(STORAGE_KEYS.session, { storeId: newStore.id });

            showToast('تم إنشاء المتجر بنجاح وتفعيل لوحة التحكم!');
            checkMerchantState();
        }

        function logoutMerchant() {
            currentMerchantStore = null;
            localStorage.removeItem(STORAGE_KEYS.session);
            checkMerchantState();
            showToast('تم الخروج من لوحة التاجر');
        }

        function addNewProduct() {
            if (!currentMerchantStore) return;

            const title = document.getElementById('productTitle').value.trim();
            const price = Number(document.getElementById('productPrice').value);
            const image = document.getElementById('productImage').value.trim();
            const desc = document.getElementById('productDesc').value.trim();

            if (title.length < 3 || !Number.isInteger(price) || price <= 0) {
                showToast('يرجى إدخال عنوان صالح للمنتج وسعر صحيح');
                return;
            }

            if (image && !isSafeImageUrl(image)) {
                showToast('رابط الصورة غير صالح');
                return;
            }

            const newProd = {
                id: 'p-' + crypto.randomUUID(),
                title: title,
                price: price,
                image: image || 'https://images.unsplash.com/photo-1523275335684-37898b6baf30?auto=format&fit=crop&w=600&q=80',
                desc: desc || 'منتج جديد عالي الجودة متوفر للطلب الفوري.',
                stock: 10,
                status: 'approved',
                active: true
            };

            const store = dbStores.find(s => s.id === currentMerchantStore.id);
            if (store) {
                store.products.unshift(newProd);
                currentMerchantStore = store;
                writeJson(STORAGE_KEYS.stores, dbStores);

                document.getElementById('productTitle').value = '';
                document.getElementById('productPrice').value = '';
                document.getElementById('productImage').value = '';
                document.getElementById('productDesc').value = '';

                showToast('تم نشر السلعة بنجاح وإتاحتها للزبائن!');
                renderMerchantProducts();
            }
        }

        function deleteProduct(prodId) {
            const store = dbStores.find(s => s.id === currentMerchantStore?.id);
            if (store) {
                store.products = store.products.filter(p => p.id !== prodId);
                currentMerchantStore = store;
                writeJson(STORAGE_KEYS.stores, dbStores);
                renderMerchantProducts();
                showToast('تم حذف المنتج بنجاح');
            }
        }
